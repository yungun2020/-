        function switchTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            const targetTab = document.getElementById(`tab-${tabName}`);
            if (targetTab) targetTab.classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('text-amber-700', 'font-bold', 'bg-amber-50');
                btn.classList.add('text-stone-600');
            });
            const activeNav = document.getElementById(`nav-${tabName}`);
            if (activeNav) {
                activeNav.classList.add('text-amber-700', 'font-bold', 'bg-amber-50');
                activeNav.classList.remove('text-stone-600');
            }

            // Bottom Navigation active highlight
            ['home', 'business', 'clubs', 'jobs', 'chat'].forEach(t => {
                const bEl = document.getElementById(`b-nav-${t}`);
                if (bEl) {
                    if (t === tabName) {
                        bEl.classList.remove('text-stone-400');
                        bEl.classList.add('text-amber-600', 'font-bold');
                    } else {
                        bEl.classList.add('text-stone-400');
                        bEl.classList.remove('text-amber-600', 'font-bold');
                    }
                }
            });

            if (tabName === 'home') {
                const input = document.getElementById('quickSearchInput');
                if (input) input.value = '';
                renderBusiness(businessList);
            }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

                    <span class="px-2 py-0.5 bg-amber-600 text-white font-bold rounded w-max text-[10px]">${j.type}</span>
                    <h4 class="font-bold text-stone-900 text-sm">${j.title}</h4>
                    <p class="text-stone-600">${j.desc}</p>
                    <button onclick="alert('담당자에게 연결됩니다.')" class="w-full py-2 bg-stone-100 text-stone-700 border border-stone-300 font-bold rounded-xl hover:bg-stone-200">상세 문의</button>
                </div>
            `).join('');
        }

        function renderChatRooms() {
            const container = document.getElementById('chatRoomList');
            if (!container) return;
            container.innerHTML = chatRooms.map(r => `
                <div onclick="currentRoomId=${r.id}; renderChatMessages();" class="p-2.5 rounded-xl cursor-pointer transition ${r.id === currentRoomId ? 'bg-amber-50 border border-amber-300' : 'bg-stone-50 hover:bg-stone-100'}">
                    <div class="flex justify-between items-center text-xs"><strong class="text-stone-900">${r.title}</strong><span class="text-[10px] text-amber-600">${r.count}명</span></div>
                </div>
            `).join('');
        }

        function renderChatMessages() {
            const container = document.getElementById('chatMessages');
            if (!container) return;
            container.innerHTML = chatMessages.map(m => `
                <div class="flex flex-col ${m.isMe ? 'items-end' : 'items-start'} space-y-1 text-xs">
                    <span class="text-[10px] text-stone-500">${m.sender} &bull; ${m.time}</span>
                    <div class="p-2.5 rounded-xl max-w-[80%] ${m.isMe ? 'bg-amber-600 text-white' : 'bg-white text-stone-800 border border-stone-200'}">${m.text}</div>
                </div>
            `).join('');
            container.scrollTop = container.scrollHeight;
        }

        function handleSendChatMessage(e) {
            e.preventDefault();
            const input = document.getElementById('chatInput');
            if (!input || !input.value.trim()) return;
            chatMessages.push({ id: chatMessages.length + 1, roomId: currentRoomId, sender: "나", text: input.value.trim(), time: "방금 전", isMe: true });
            input.value = '';
            renderChatMessages();
        }

        function filterClubs(cat, btn) {
            document.querySelectorAll('.club-cat-btn').forEach(b => {
                b.classList.remove('bg-amber-600', 'text-white');
                b.classList.add('bg-stone-100', 'text-stone-700');
            });
            btn.classList.add('bg-amber-600', 'text-white');
            renderClubs();
        }

            renderEvents();
            renderJobs();
            renderChatRooms();
            renderChatMessages();
        };
